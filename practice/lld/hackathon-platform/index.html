HackathonPlatformDemo.java

Question

Design a coding-contest platform.

  1. Add problems to a question library: description, tags, difficulty, score.
  2. Contestants register with a name and department.
  3. Fetch problems filtered by difficulty or tag and sorted by score, extensible to new filters and sorts.
  4. A contestant solves a problem (with the time taken); list the problems a contestant has solved.
  5. For a problem: how many users solved it and the average time taken.
  6. Scoring is pluggable: a fixed score, or a score adjusted by solve time.
  7. Return the current leader.
  8. Top N most-solved problems for a tag.
  9. Extension: after a solve, recommend 5 problems with similar tags.

Design: repositories + services, Strategy for scoring and recommendations, Predicate / Comparator for filters, custom exceptions, unit tests in src/test. Run HackathonPlatformDemo; it prints each requirement in order.

        return contestantRepository.findById(contestantId)
                .orElseThrow(() -> new EntityNotFoundException("Contestant not found: " + contestantId));

package com.practice.phonepe.interview.service;

import com.practice.phonepe.interview.entities.Contestant;
import com.practice.phonepe.interview.entities.Problem;
import com.practice.phonepe.interview.entities.Submission;
import com.practice.phonepe.interview.exception.AlreadySolvedException;
import com.practice.phonepe.interview.exception.EntityNotFoundException;
import com.practice.phonepe.interview.repository.ContestantRepository;
import com.practice.phonepe.interview.repository.ProblemRepository;
import com.practice.phonepe.interview.repository.SubmissionRepository;
import com.practice.phonepe.interview.strategy.RecommendationStrategy;
import com.practice.phonepe.interview.strategy.ScoringStrategy;

import java.security.KeyPair;
import java.util.*;
import java.util.stream.Collectors;

class Pair {
    int key;
    Contestant contestant;
    public Pair(int key , Contestant contestant) {
        this.key = key;
        this.contestant = contestant;
    }
}

public class ContestService {

    private static final int RECOMMENDATION_COUNT = 5;

    private final ContestantRepository contestantRepository;
    private final ProblemRepository problemRepository;
    private final SubmissionRepository submissionRepository;
    private final ScoringStrategy scoringStrategy;
    private final RecommendationStrategy recommendationStrategy;
    private final PriorityQueue<Pair> leaderBoard;

    public ContestService(ContestantRepository contestantRepository,
                           ProblemRepository problemRepository,
                           SubmissionRepository submissionRepository,
                           ScoringStrategy scoringStrategy,
                           RecommendationStrategy recommendationStrategy) {
        this.contestantRepository = contestantRepository;
        this.problemRepository = problemRepository;
        this.submissionRepository = submissionRepository;
        this.scoringStrategy = scoringStrategy;
        this.recommendationStrategy = recommendationStrategy;
        this.leaderBoard = new PriorityQueue<>();
    }

    public SolveResult solve(String contestantId, String problemId, int timeTakenSeconds) {
        Contestant contestant = contestantRepository.findById(contestantId)
                .orElseThrow(() -> new EntityNotFoundException("Contestant not found: " + contestantId));
        Problem problem = problemRepository.findById(problemId)
                .orElseThrow(() -> new EntityNotFoundException("Problem not found: " + problemId));

        Set<String> solvedProblemIds = submissionRepository.findByContestant(contestantId).stream()
                .map(Submission::getProblemId)
                .collect(Collectors.toSet());
        if (solvedProblemIds.contains(problemId)) {
            throw new AlreadySolvedException(contestant.getName() + " has already solved this problem");
        }

        int scoreAwarded = scoringStrategy.calculateScore(problem, timeTakenSeconds);
        submissionRepository.save(new Submission(contestantId, problemId, timeTakenSeconds, scoreAwarded));
        //
        Pair pair = new Pair(scoreAwarded, contestant);
        leaderBoard.add(pair);

        List<Problem> candidates = problemRepository.findAll().stream()
                .filter(candidate -> !candidate.getId().equals(problemId))
                .filter(candidate -> !solvedProblemIds.contains(candidate.getId()))
                .collect(Collectors.toList());
        List<Problem> recommendations = recommendationStrategy.recommend(problem, candidates, RECOMMENDATION_COUNT);
        // M - contestant , N - problem
        // O(log(M)) + O(N)
        return new SolveResult(scoreAwarded, recommendations);
    }

    public Optional<Contestant> getLeader() {
        if (submissionRepository.findAll().isEmpty()) {
            return Optional.empty();
        }

        Map<String, Integer> totalScoreByContestant = submissionRepository.findAll().stream()
                .collect(Collectors.groupingBy(Submission::getContestantId, Collectors.summingInt(Submission::getScoreAwarded)));

        // Assumption: ties broken by earliest-registered contestant
        Contestant leader = null;
        int bestScore = Integer.MIN_VALUE;
        for (Contestant contestant : contestantRepository.findAll()) {
            int score = totalScoreByContestant.getOrDefault(contestant.getId(), 0);
            if (score > bestScore) {
                bestScore = score;
                leader = contestant;
            }
        }
        return Optional.ofNullable(leader);
    }


}

package com.practice.phonepe.interview.service;

import com.practice.phonepe.interview.entities.Problem;
import com.practice.phonepe.interview.entities.Submission;
import com.practice.phonepe.interview.exception.EntityNotFoundException;
import com.practice.phonepe.interview.repository.ProblemRepository;
import com.practice.phonepe.interview.repository.SubmissionRepository;

import java.util.Comparator;
import java.util.List;
import java.util.function.Predicate;
import java.util.stream.Collectors;

public class ProblemService {

    private final ProblemRepository problemRepository;
    private final SubmissionRepository submissionRepository;

    public ProblemService(ProblemRepository problemRepository, SubmissionRepository submissionRepository) {
        this.problemRepository = problemRepository;
        this.submissionRepository = submissionRepository;
    }

    public void addProblem(Problem problem) {
        problemRepository.save(problem);
    }

    public List<Problem> fetchProblems(Predicate<Problem> filter, Comparator<Problem> sortBy) {
        return problemRepository.findAll().stream()
                .filter(filter)
                .sorted(sortBy)
                .collect(Collectors.toList());
    }

    public ProblemStats getProblemStats(String problemId) {
        requireProblem(problemId);
        List<Submission> submissions = submissionRepository.findByProblem(problemId);
        double averageTime = submissions.stream()
                .mapToInt(Submission::getTimeTakenSeconds)
                .average()
                .orElse(0);
        return new ProblemStats(submissions.size(), averageTime);
    }

    // "Liked" isn't defined anywhere in the spec beyond solving, so it's treated as solve count.
    public List<Problem> getTopNProblems(String tag, int n) {
        return problemRepository.findAll().stream()
                .filter(problem -> problem.getTags().contains(tag))
                .sorted(Comparator.comparingInt((Problem problem) -> submissionRepository.findByProblem(problem.getId()).size()).reversed())
                .limit(n)
                .collect(Collectors.toList());
    }

    private Problem requireProblem(String problemId) {
        return problemRepository.findById(problemId)
                .orElseThrow(() -> new EntityNotFoundException("Problem not found: " + problemId));
    }
}

package com.practice.phonepe.interview.strategy;

import com.practice.phonepe.interview.entities.Problem;

import java.util.Comparator;
import java.util.HashSet;
import java.util.List;
import java.util.Set;
import java.util.stream.Collectors;

public class SimilarTagRecommendationStrategy implements RecommendationStrategy {

    @Override
    public List<Problem> recommend(Problem justSolved, List<Problem> candidates, int count) {
        return candidates.stream()
                .sorted(Comparator.comparingInt((Problem candidate) -> sharedTagCount(justSolved, candidate)).reversed())
                .limit(count)
                .collect(Collectors.toList());
    }

    private int sharedTagCount(Problem a, Problem b) {
        Set<String> shared = new HashSet<>(a.getTags());
        shared.retainAll(b.getTags());
        return shared.size();
    }
}

package com.practice.phonepe.interview.strategy;

import com.practice.phonepe.interview.entities.Problem;

public class TimeAdjustedScoreStrategy implements ScoringStrategy {

    private final int fastSolveThresholdSeconds;

    public TimeAdjustedScoreStrategy(int fastSolveThresholdSeconds) {
        this.fastSolveThresholdSeconds = fastSolveThresholdSeconds;
    }

    @Override
    public int calculateScore(Problem problem, int timeTakenSeconds) {
        // Assumption: no exact formula was specified. Full score within the threshold, then a
        // 10% penalty per additional threshold interval, floored at half the base score.
        if (timeTakenSeconds <= fastSolveThresholdSeconds) {
            return problem.getBaseScore();
        }
        int intervalsLate = (timeTakenSeconds - fastSolveThresholdSeconds) / fastSolveThresholdSeconds;
        double penalty = Math.min(0.5, intervalsLate * 0.1);
        return (int) Math.round(problem.getBaseScore() * (1 - penalty));
    }
}

        System.out.println("  TimeAdjustedScoreStrategy -> " + scoringStrategy.calculateScore(mergeIntervals, slowSolveTime));

        requirement(8, "return the current leader of the contest");
        Optional<Contestant> leader = contestService.getLeader();
        leader.ifPresent(c -> System.out.println(c.getName() + " (" + c.getDepartment() + ") is leading"));

        requirement(9, "top N liked problems of a given tag (\"liked\" = solve count)");
        System.out.println("top 10 tagged 'array' (only 3 exist, so all 3 come back):");
        printProblems(problemService.getTopNProblems("array", 10));

        extension("on solving a problem, return top 5 recommended problems by similar tags");
        for (SolveResult solve : solves) {
            if (!solve.getRecommendedProblems().isEmpty()) {
                System.out.println("  recommended: " + describe(solve.getRecommendedProblems()));
            }
        }

        note("exceptions handled gracefully, errors printed to the console");
        try {
            contestService.solve("no-such-contestant", twoSum.getId(), 100);
        } catch (EntityNotFoundException e) {
            System.err.println("ERROR: " + e.getMessage());
        }
        try {
            contestService.solve(aayushi.getId(), "no-such-problem", 100);
        } catch (EntityNotFoundException e) {
            System.err.println("ERROR: " + e.getMessage());
        }
        try {
            contestService.solve(aayushi.getId(), twoSum.getId(), 150); // already solved above
        } catch (AlreadySolvedException e) {
            System.err.println("ERROR: " + e.getMessage());
        }
    }

    private static SolveResult solveAndPrint(ContestService contestService, Contestant contestant, Problem problem, int timeTakenSeconds) {
        SolveResult result = contestService.solve(contestant.getId(), problem.getId(), timeTakenSeconds);
        System.out.println(contestant.getName() + " solved '" + problem.getDescription()
                + "' in " + timeTakenSeconds + "s -> scored " + result.getScoreAwarded());
        return result;
    }

    private static void requirement(int number, String description) {
        System.out.println();
        System.out.println("=== [Mandatory " + number + "] " + description + " ===");
    }

    private static void extension(String description) {
        System.out.println();
        System.out.println("=== [Extension] " + description + " ===");
    }

    private static void note(String description) {
        System.out.println();
        System.out.println("=== [" + description + "] ===");
    }

    private static void printProblems(List<Problem> problems) {
        if (problems.isEmpty()) {
            System.out.println("  (none)");
            return;
        }
        for (Problem problem : problems) {
            System.out.printf("  - %-28s tags=%-24s difficulty=%-6s score=%d%n",
                    problem.getDescription(), problem.getTags(), problem.getDifficulty(), problem.getBaseScore());
        }
    }

    private static String describe(List<Problem> problems) {
        StringBuilder builder = new StringBuilder();
        for (Problem problem : problems) {
            if (builder.length() > 0) {
                builder.append(", ");
            }
            builder.append(problem.getDescription());
        }
        return builder.toString();
    }
