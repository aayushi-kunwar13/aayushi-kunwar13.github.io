package com.practice.phonepe.questions.notificationsystemv2.channel;

import com.practice.phonepe.questions.notificationsystemv2.entities.ChannelType;

import java.util.Map;

public class NotificationChannelFactory {
    private final Map<ChannelType, NotificationChannel> channels;

    public NotificationChannelFactory(Map<ChannelType, NotificationChannel> channels) {
        this.channels = channels;
    }

    public NotificationChannel get(ChannelType type) {
        NotificationChannel channel = channels.get(type);
        if (channel == null) {
            throw new IllegalArgumentException("No channel registered for " + type);
        }
        return channel;
    }
}

package com.practice.phonepe.questions.notificationsystemv2.channel;

import com.practice.phonepe.questions.notificationsystemv2.entities.ChannelType;
import com.practice.phonepe.questions.notificationsystemv2.entities.Notification;
import com.practice.phonepe.questions.notificationsystemv2.entities.User;

/**
 * Wraps any channel and retries on failure. Keeps retry logic out of every
 * individual channel implementation - a channel just knows how to send once.
 */
public class RetryingNotificationChannel implements NotificationChannel {
    private final NotificationChannel delegate;
    private final int maxAttempts;

    public RetryingNotificationChannel(NotificationChannel delegate, int maxAttempts) {
        this.delegate = delegate;
        this.maxAttempts = maxAttempts;
    }

    @Override
    public void send(User user, Notification notification) {
        for (int attempt = 1; attempt <= maxAttempts; attempt++) {
            try {
                delegate.send(user, notification);
                return;
            } catch (RuntimeException e) {
                System.out.println("[RETRY] " + delegate.getType() + " attempt " + attempt + "/" + maxAttempts
                        + " failed: " + e.getMessage());
                if (attempt == maxAttempts) {
                    throw e;
                }
            }
        }
    }

    @Override
    public ChannelType getType() {
        return delegate.getType();
    }
}

package com.practice.phonepe.questions.notificationsystemv2.observer;

import com.practice.phonepe.questions.notificationsystemv2.entities.ChannelType;
import com.practice.phonepe.questions.notificationsystemv2.entities.User;

import java.util.List;
import java.util.concurrent.CopyOnWriteArrayList;

/**
 * The Subject. Owns nothing but the observer list - dispatch logic stays in
 * NotificationService, so adding a new observer never touches it.
 */
public class NotificationEventPublisher {
    private final List<NotificationObserver> observers = new CopyOnWriteArrayList<>();

    public void subscribe(NotificationObserver observer) {
        observers.add(observer);
    }

    public void publish(User user, ChannelType type, boolean success, String detail) {
        for (NotificationObserver observer : observers) {
            observer.onResult(user, type, success, detail);
        }
    }
}

package com.practice.phonepe.questions.notificationsystemv2.service;

import com.practice.phonepe.questions.notificationsystemv2.channel.NotificationChannelFactory;
import com.practice.phonepe.questions.notificationsystemv2.entities.ChannelType;
import com.practice.phonepe.questions.notificationsystemv2.entities.Notification;
import com.practice.phonepe.questions.notificationsystemv2.entities.NotificationStatus;
import com.practice.phonepe.questions.notificationsystemv2.entities.User;
import com.practice.phonepe.questions.notificationsystemv2.observer.NotificationEventPublisher;

import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;

public class NotificationService {
    private final NotificationChannelFactory factory;
    private final NotificationEventPublisher publisher;
    private final ExecutorService executor = Executors.newFixedThreadPool(4);
    private final Map<String, Map<ChannelType, NotificationStatus>> statusByNotification = new ConcurrentHashMap<>();

    public NotificationService(NotificationChannelFactory factory, NotificationEventPublisher publisher) {
        this.factory = factory;
        this.publisher = publisher;
    }

    public void notifyUser(User user, Notification notification) {
        for (ChannelType type : user.getPreferredChannels()) {
            executor.submit(() -> {
                try {
                    factory.get(type).send(user, notification);
                    recordStatus(notification, type, NotificationStatus.SENT);
                    publisher.publish(user, type, true, null);
                } catch (RuntimeException e) {
                    recordStatus(notification, type, NotificationStatus.FAILED);
                    publisher.publish(user, type, false, e.getMessage());
                }
            });
        }
    }

    private void recordStatus(Notification notification, ChannelType type, NotificationStatus status) {
        statusByNotification
                .computeIfAbsent(notification.getId(), id -> new ConcurrentHashMap<>())
                .put(type, status);
    }

    /** Per-channel outcome for a given notification - each channel gets its own slot. */
    public Map<ChannelType, NotificationStatus> getStatus(String notificationId) {
        return statusByNotification.getOrDefault(notificationId, Map.of());
    }

    public void shutdown() {
        executor.shutdown();
        try {
            executor.awaitTermination(5, TimeUnit.SECONDS);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
    }
}

package com.practice.phonepe.questions.notificationsystemv2;

import com.practice.phonepe.questions.notificationsystemv2.channel.EmailChannel;
import com.practice.phonepe.questions.notificationsystemv2.channel.NotificationChannelFactory;
import com.practice.phonepe.questions.notificationsystemv2.channel.RetryingNotificationChannel;
import com.practice.phonepe.questions.notificationsystemv2.channel.SmsChannel;
import com.practice.phonepe.questions.notificationsystemv2.entities.ChannelType;
import com.practice.phonepe.questions.notificationsystemv2.entities.Notification;
import com.practice.phonepe.questions.notificationsystemv2.entities.User;
import com.practice.phonepe.questions.notificationsystemv2.observer.AuditLogObserver;
import com.practice.phonepe.questions.notificationsystemv2.observer.NotificationEventPublisher;
import com.practice.phonepe.questions.notificationsystemv2.service.NotificationService;

import java.util.EnumSet;
import java.util.Map;

public class Demo {
    public static void main(String[] args) {
        NotificationChannelFactory factory = new NotificationChannelFactory(Map.of(
                ChannelType.EMAIL, new RetryingNotificationChannel(new EmailChannel(), 3),
                ChannelType.SMS, new SmsChannel()));
        NotificationEventPublisher publisher = new NotificationEventPublisher();
        publisher.subscribe(new AuditLogObserver());
        NotificationService service = new NotificationService(factory, publisher);

        User alice = new User("u1", "Alice", "alice@example.com", "+1-555-0100",
                EnumSet.of(ChannelType.EMAIL, ChannelType.SMS));
        User bob = new User("u2", "Bob", "bob@example.com", "+1-555-0200",
                EnumSet.of(ChannelType.SMS));

        Notification forAlice = new Notification("Order Update", "Your order has shipped!");
        Notification forBob = new Notification("Order Update", "Your order has shipped!");
        service.notifyUser(alice, forAlice);
        service.notifyUser(bob, forBob);

        service.shutdown();
        System.out.println("Alice's notification status per channel: " + service.getStatus(forAlice.getId()));
        System.out.println("Bob's notification status per channel: " + service.getStatus(forBob.getId()));
    }
}
