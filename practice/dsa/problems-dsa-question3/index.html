Question

Topic: Hashing / Caching · Difficulty: Medium

Requests arrive in order, each with an optional idempotency key and a value (the response it would produce). Return the response sent for every request.

Part 1: replay by key

  • No key → process it and return its own value.
  • First time a key is seen → process it and remember the response.
  • Key seen before → return the stored response (a replay), not the new value.

Part 2: keys expire

Each request also has a timestamp, and stored keys expire after ttl. A request after storedTime + ttl is processed fresh and replaces the stored response.

Part 3: conflicting retries

Within the TTL, a request with the same key but a different value is a client error: return -1. The same value is a normal retry.

Approach: HashMap<key, (response, expiresAt)>, checked on every request.

 * Problem: Idempotent Request Replay
 * Difficulty: Medium
 * Topic: Hashing / Caching
 * Question: Replay stored responses for repeated idempotency keys, then with key expiry (TTL), then rejecting conflicting retries. Full statement in README.md.
 */
package dsa.question3;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;

public class IdempotentReplayPart1 {

    static class Request {
        String key;
        int value;
        public Request(String key, int value) {
            this.key = key;
            this.value = value;
        }
    }

    public static List<Integer> inputRequest(List<Request> input) {
        HashMap<String , Integer> requestedKey = new HashMap<>();
        List<Integer> output = new ArrayList<>();
        for(int i=0; i<input.size(); i++) {
            String key = input.get(i).key;
            int value = input.get(i).value;
            if(key == null || key.equals("")) {
                output.add(value);
            }else if(requestedKey.containsKey(key)) {
                output.add(requestedKey.get(key));
            }else{
                output.add(value);
                requestedKey.put(key, value);
            }
        }
        return output;
    }
}

package dsa.question3;

import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;

public class IdempotentReplayPart2 {

    static class Request {
        String key;
        int value;
        long time;
        public Request(String key, int value, long time) {
            this.key = key;
            this.value = value;
            this.time = time;
        }
    }

    static class Pair{
        int value;
        long expireTime;
        public Pair(int value, long expireTime) {
            this.value = value;
            this.expireTime = expireTime;
        }
    }

    public static List<Integer> inputRequest(List<Request> processRequest, long ttl) {
        HashMap<String , Pair> store = new HashMap<>();
        List<Integer> output = new ArrayList<>();
        for(int i=0; i<processRequest.size(); i++) {

            String key = processRequest.get(i).key;
            int value = processRequest.get(i).value;
            long time = processRequest.get(i).time;

            if(key == null || key.isEmpty()){
                output.add(value);
            } else if(store.containsKey(key)) {
                Pair lastValue = store.get(key);
                if(time > lastValue.expireTime) {
                    output.add(value);
                    store.put(key , new Pair(value, time+ttl));
                }else{
                    output.add(lastValue.value);
                }

            }else {
                output.add(value);
                store.put(key , new Pair(value, time+ttl));
            }
        }
        return output;
    }
}

                output.add(value);

                if(time > lastValue.expireTime) {
                    output.add(value);
                    store.put(key , new Pair(value, time+ttl));

                    output.add(-1);
                } else{
                    output.add(value);
                }

            }else {
                output.add(value);
                store.put(key , new Pair(value, time+ttl));
            }
        }
        return output;
