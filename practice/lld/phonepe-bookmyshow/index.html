Question

Design the backend for booking movie tickets.

  • Admins add movies, theatres and screens (seat rows with a type: normal, premium, recliner) and create shows.
  • Users search shows for a movie in a city and see seat availability.
  • A user picks seats. The seats are locked for a short time (30s here) while they pay, and other users can’t take them.
  • Pay by card or UPI. On success the booking is confirmed; if the lock expires, the seats are released.
  • Price depends on seat type.
  • Concurrency: many users racing for the same seat; exactly one may win.

Design: repositories + services, SeatLockManager (per-seat locks with expiry), Strategy for pricing and payment, a factory for payment methods, and custom exceptions. Run BookMyShowDemo (includes a 10-thread race for one seat).

package com.practice.phonepe.questions.bookmyshow.entities;

/**
 * Per-show mutable booking state for a physical {@link Seat}. The same seat is reused across
 * many shows, so availability/locking is tracked here instead of on the seat itself.
 */
public class ShowSeat {

    private final String showId;
    private final Seat seat;
    private final double price;

    private SeatStatus status;
    private String lockedByUserId;
    private long lockExpiryEpochMillis;

    public ShowSeat(String showId, Seat seat, double price) {
        this.showId = showId;
        this.seat = seat;
        this.price = price;
        this.status = SeatStatus.AVAILABLE;
    }

    public String getId() {
        return showId + "_" + seat.getId();
    }

    public String getShowId() {
        return showId;
    }

    public Seat getSeat() {
        return seat;
    }

    public double getPrice() {
        return price;
    }

    public synchronized SeatStatus getStatus() {
        return status;
    }

    public synchronized String getLockedByUserId() {
        return lockedByUserId;
    }

    /** Attempts to move AVAILABLE -> LOCKED for the given user. Self-heals stale locks that outlived their TTL. */
    public synchronized boolean tryLock(String userId, long ttlMillis, long nowEpochMillis) {
        expireIfNeeded(nowEpochMillis);
        if (status != SeatStatus.AVAILABLE) {
            return false;
        }
        status = SeatStatus.LOCKED;
        lockedByUserId = userId;
        lockExpiryEpochMillis = nowEpochMillis + ttlMillis;
        return true;
    }

    /** Moves LOCKED -> BOOKED, only if still held by the same user and not expired. */
    public synchronized boolean confirmBooking(String userId, long nowEpochMillis) {
        expireIfNeeded(nowEpochMillis);
        if (status != SeatStatus.LOCKED || !lockedByUserId.equals(userId)) {
            return false;
        }
        status = SeatStatus.BOOKED;
        return true;
    }

    /** Releases a LOCKED seat held by the given user back to AVAILABLE (cancellation / payment failure). */
    public synchronized void release(String userId) {
        if (status == SeatStatus.LOCKED && userId.equals(lockedByUserId)) {
            status = SeatStatus.AVAILABLE;
            lockedByUserId = null;
        }
    }

    /** Called by the background sweep; releases an expired lock back to AVAILABLE. Returns true if it did. */
    public synchronized boolean expireIfNeeded(long nowEpochMillis) {
        if (status == SeatStatus.LOCKED && nowEpochMillis >= lockExpiryEpochMillis) {
            status = SeatStatus.AVAILABLE;
            lockedByUserId = null;
            return true;
        }
        return false;
    }
}

package com.practice.phonepe.questions.bookmyshow.lock;

import com.practice.phonepe.questions.bookmyshow.entities.ShowSeat;
import com.practice.phonepe.questions.bookmyshow.exception.SeatUnavailableException;

import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.Executors;
import java.util.concurrent.ScheduledExecutorService;
import java.util.concurrent.TimeUnit;

/**
 * Coordinates temporary seat holds while a user is checking out. A seat picked by one user is
 * LOCKED for a short TTL so nobody else can grab it; if checkout doesn't complete in time, a
 * background sweep releases it automatically so it becomes bookable again. This is what keeps
 * seat availability correct under concurrent bookings without holding the lock forever.
 */
public class SeatLockManager {

    private static final long DEFAULT_TTL_MILLIS = TimeUnit.MINUTES.toMillis(5);
    private static final long SWEEP_INTERVAL_MILLIS = TimeUnit.SECONDS.toMillis(1);

    private final long lockTtlMillis;
    private final ConcurrentHashMap<String, ShowSeat> activeLocks = new ConcurrentHashMap<>();
    private final ScheduledExecutorService scheduler = Executors.newSingleThreadScheduledExecutor(r -> {
        Thread t = new Thread(r, "seat-lock-sweeper");
        t.setDaemon(true);
        return t;
    });

    public SeatLockManager() {
        this(DEFAULT_TTL_MILLIS);
    }

    public SeatLockManager(long lockTtlMillis) {
        this.lockTtlMillis = lockTtlMillis;
        scheduler.scheduleAtFixedRate(this::sweepExpiredLocks, SWEEP_INTERVAL_MILLIS, SWEEP_INTERVAL_MILLIS, TimeUnit.MILLISECONDS);
    }

    /**
     * Locks every seat for the given user, or none at all. If any requested seat is already
     * taken, whatever was locked earlier in this call is rolled back before throwing.
     */
    public List<ShowSeat> lockSeats(String userId, List<ShowSeat> seats) {
        long now = System.currentTimeMillis();
        List<ShowSeat> locked = new ArrayList<>();
        for (ShowSeat seat : seats) {
            if (seat.tryLock(userId, lockTtlMillis, now)) {
                locked.add(seat);
                activeLocks.put(seat.getId(), seat);
            } else {
                releaseSeats(userId, locked);
                throw new SeatUnavailableException("Seat " + seat.getSeat().getLabel() + " is no longer available");
            }
        }
        return locked;
    }

    /** Moves held seats from LOCKED to BOOKED once payment succeeds. */
    public void confirmSeats(String userId, List<ShowSeat> seats) {
        long now = System.currentTimeMillis();
        for (ShowSeat seat : seats) {
            seat.confirmBooking(userId, now);
            activeLocks.remove(seat.getId());
        }
    }

    /** Releases held seats back to AVAILABLE, e.g. on cancellation or payment failure. */
    public void releaseSeats(String userId, List<ShowSeat> seats) {
        for (ShowSeat seat : seats) {
            seat.release(userId);
            activeLocks.remove(seat.getId());
        }
    }

    private void sweepExpiredLocks() {
        long now = System.currentTimeMillis();
        activeLocks.forEach((id, seat) -> {
            if (seat.expireIfNeeded(now)) {
                activeLocks.remove(id);
            }
        });
    }

    public void shutdown() {
        scheduler.shutdownNow();
    }
}

package com.practice.phonepe.questions.bookmyshow.payment;

import java.util.EnumMap;
import java.util.Map;

/**
 * Resolves the {@link PaymentStrategy} for a {@link PaymentMethod} so callers depend only on the
 * method the user picked, never on a concrete strategy class.
 */
public class PaymentStrategyFactory {

    private final Map<PaymentMethod, PaymentStrategy> strategiesByMethod = new EnumMap<>(PaymentMethod.class);

    public PaymentStrategyFactory() {
        strategiesByMethod.put(PaymentMethod.CARD, new CardPaymentStrategy());
        strategiesByMethod.put(PaymentMethod.UPI, new UpiPaymentStrategy());
    }

    public PaymentStrategy getStrategy(PaymentMethod method) {
        PaymentStrategy strategy = strategiesByMethod.get(method);
        if (strategy == null) {
            throw new IllegalArgumentException("No payment strategy registered for method: " + method);
        }
        return strategy;
    }

    public void registerStrategy(PaymentMethod method, PaymentStrategy strategy) {
        strategiesByMethod.put(method, strategy);
    }
}

    }

                .filter(show -> show.getMovieId().equals(movieId))
                .filter(show -> theatreRepository.findById(show.getTheatreId())
                        .map(Theatre::getCity)
                        .map(theatreCity -> theatreCity.equalsIgnoreCase(city))
                        .orElse(false))
                .collect(Collectors.toList());

import com.practice.phonepe.questions.bookmyshow.entities.Booking;
import com.practice.phonepe.questions.bookmyshow.entities.BookingStatus;
import com.practice.phonepe.questions.bookmyshow.entities.Payment;
import com.practice.phonepe.questions.bookmyshow.entities.PaymentStatus;
import com.practice.phonepe.questions.bookmyshow.entities.Show;
import com.practice.phonepe.questions.bookmyshow.entities.ShowSeat;
import com.practice.phonepe.questions.bookmyshow.exception.EntityNotFoundException;
import com.practice.phonepe.questions.bookmyshow.exception.InvalidSeatLockException;
import com.practice.phonepe.questions.bookmyshow.lock.SeatLockManager;
import com.practice.phonepe.questions.bookmyshow.payment.PaymentMethod;
import com.practice.phonepe.questions.bookmyshow.payment.PaymentStrategy;
import com.practice.phonepe.questions.bookmyshow.payment.PaymentStrategyFactory;
import com.practice.phonepe.questions.bookmyshow.repository.BookingRepository;

import java.util.List;
import java.util.stream.Collectors;

/**
 * Orchestrates the booking flow: hold seats -> collect payment -> confirm or roll back.
 * Seat-level concurrency is delegated entirely to {@link SeatLockManager}; this class just
 * sequences the steps and keeps {@link Booking} state consistent with seat state.
 */
public class BookingService {

    private final BookingRepository bookingRepository;
    private final ShowService showService;
    private final SeatLockManager seatLockManager;
    private final PaymentStrategyFactory paymentStrategyFactory;

    public BookingService(BookingRepository bookingRepository, ShowService showService, SeatLockManager seatLockManager,
                           PaymentStrategyFactory paymentStrategyFactory) {
        this.bookingRepository = bookingRepository;
        this.showService = showService;
        this.seatLockManager = seatLockManager;
        this.paymentStrategyFactory = paymentStrategyFactory;
    }

    /** Step 1: hold the requested seats for this user and create a pending booking. */
    public Booking initiateBooking(String userId, String showId, List<String> seatIds) {
        Show show = showService.getShow(showId);
        List<ShowSeat> requestedSeats = seatIds.stream()
                .map(seatId -> {
                    ShowSeat showSeat = show.getShowSeatsBySeatId().get(seatId);
                    if (showSeat == null) {
                        throw new EntityNotFoundException("Seat not found on show: " + seatId);
                    }
                    return showSeat;
                })
                .collect(Collectors.toList());

        List<ShowSeat> lockedSeats = seatLockManager.lockSeats(userId, requestedSeats);
        double totalAmount = lockedSeats.stream().mapToDouble(ShowSeat::getPrice).sum();

        Booking booking = new Booking(userId, showId, seatIds, totalAmount);
        bookingRepository.save(booking);
        return booking;
    }

    /** Step 2: charge the user and, on success, permanently claim the seats. */
    public Booking confirmBooking(String bookingId, PaymentMethod paymentMethod) {
        Booking booking = getBooking(bookingId);
        if (booking.getStatus() != BookingStatus.CREATED) {
            throw new InvalidSeatLockException("Booking " + bookingId + " is not awaiting payment");
        }

        Show show = showService.getShow(booking.getShowId());
        List<ShowSeat> seats = booking.getSeatIds().stream()
                .map(seatId -> show.getShowSeatsBySeatId().get(seatId))
                .collect(Collectors.toList());

        PaymentStrategy paymentStrategy = paymentStrategyFactory.getStrategy(paymentMethod);
        boolean paymentSucceeded = paymentStrategy.pay(booking.getUserId(), booking.getTotalAmount());
        if (!paymentSucceeded) {
            seatLockManager.releaseSeats(booking.getUserId(), seats);
            booking.setStatus(BookingStatus.CANCELLED);
            booking.setPayment(new Payment(bookingId, booking.getTotalAmount(), paymentMethod.name(), PaymentStatus.FAILED));
            bookingRepository.save(booking);
            return booking;
        }

        boolean allConfirmed = seats.stream()
                .allMatch(seat -> seat.confirmBooking(booking.getUserId(), System.currentTimeMillis()));
        if (!allConfirmed) {
            // Hold expired before payment finished; refund path would go here in a real system.
            seatLockManager.releaseSeats(booking.getUserId(), seats);
            booking.setStatus(BookingStatus.EXPIRED);
            bookingRepository.save(booking);
            throw new InvalidSeatLockException("Seat lock expired before payment could be confirmed for booking " + bookingId);
        }

        booking.setStatus(BookingStatus.CONFIRMED);
        booking.setPayment(new Payment(bookingId, booking.getTotalAmount(), paymentMethod.name(), PaymentStatus.SUCCESS));
        bookingRepository.save(booking);
        return booking;
    }

    public void cancelBooking(String bookingId) {
        Booking booking = getBooking(bookingId);
        Show show = showService.getShow(booking.getShowId());
        List<ShowSeat> seats = booking.getSeatIds().stream()
                .map(seatId -> show.getShowSeatsBySeatId().get(seatId))
                .collect(Collectors.toList());
        seatLockManager.releaseSeats(booking.getUserId(), seats);
        booking.setStatus(BookingStatus.CANCELLED);
        bookingRepository.save(booking);
    }

    public Booking getBooking(String bookingId) {
        return bookingRepository.findById(bookingId)
                .orElseThrow(() -> new EntityNotFoundException("Booking not found: " + bookingId));
    }
}

public class BookMyShowDemo {

    public static void main(String[] args) throws InterruptedException {
        // --- wiring ---
        MovieRepository movieRepository = new InMemoryMovieRepository();
        TheatreRepository theatreRepository = new InMemoryTheatreRepository();
        ShowRepository showRepository = new InMemoryShowRepository(theatreRepository);
        BookingRepository bookingRepository = new InMemoryBookingRepository();

        MovieService movieService = new MovieService(movieRepository);
        TheatreService theatreService = new TheatreService(theatreRepository);
        ShowService showService = new ShowService(showRepository, theatreService, new DefaultPricingStrategy());
        SeatLockManager seatLockManager = new SeatLockManager(TimeUnit.SECONDS.toMillis(30));
        PaymentStrategyFactory paymentStrategyFactory = new PaymentStrategyFactory();
        BookingService bookingService = new BookingService(bookingRepository, showService, seatLockManager, paymentStrategyFactory);

        // --- admin sets up catalog, theatre and a show ---
        Movie movie = movieService.addMovie("Inception", "English", "Sci-Fi", 148);
        Theatre theatre = theatreService.addTheatre("PVR Forum", "Bengaluru");

        Map<String, SeatType> seatLayout = new LinkedHashMap<>();
        seatLayout.put("A", SeatType.NORMAL);
        seatLayout.put("B", SeatType.PREMIUM);
        seatLayout.put("C", SeatType.RECLINER);
        Screen screen = theatreService.addScreen(theatre.getId(), "Screen 1", seatLayout, 10);

        Show show = showService.createShow(movie.getId(), theatre.getId(), screen.getId(),
                LocalDateTime.now().plusHours(3), LocalDateTime.now().plusHours(5));

        System.out.println("Show created: " + movie + " at " + theatre.getName() + " (" + show.getId() + ")");

        // --- a user browses shows and books a couple of seats end-to-end ---
        List<Show> shows = showService.searchShows(movie.getId(), "Bengaluru");
        System.out.println("Shows found in Bengaluru: " + shows.size());

        Seat targetSeat = screen.getSeats().get(0);
        Booking booking = bookingService.initiateBooking("user-1", show.getId(), Collections.singletonList(targetSeat.getId()));
        System.out.printf("Booking %s created, amount due: %.2f%n", booking.getId(), booking.getTotalAmount());

        Booking confirmed = bookingService.confirmBooking(booking.getId(), PaymentMethod.CARD);
        System.out.println("Booking status after payment: " + confirmed.getStatus());

        // --- concurrency demo: many users race for the same seat ---
        Seat contestedSeat = screen.getSeats().get(1);
        int contenders = 10;
        ExecutorService executor = Executors.newFixedThreadPool(contenders);
        CountDownLatch startGate = new CountDownLatch(1);
        CountDownLatch doneGate = new CountDownLatch(contenders);
        AtomicInteger successCount = new AtomicInteger();
        AtomicInteger rejectedCount = new AtomicInteger();

        for (int i = 0; i < contenders; i++) {
            String userId = "contender-" + i;
            executor.submit(() -> {
                try {
                    startGate.await();
                    bookingService.initiateBooking(userId, show.getId(), Collections.singletonList(contestedSeat.getId()));
                    successCount.incrementAndGet();
                } catch (SeatUnavailableException e) {
                    rejectedCount.incrementAndGet();
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                } finally {
                    doneGate.countDown();
                }
            });
        }

        startGate.countDown();
        doneGate.await();
        executor.shutdown();

        System.out.printf("Seat %s contested by %d users -> %d locked it, %d were rejected%n",
                contestedSeat.getLabel(), contenders, successCount.get(), rejectedCount.get());

        seatLockManager.shutdown();
    }
